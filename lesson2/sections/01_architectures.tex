\begin{frame}{Everything in the prompt}
  \begin{center}
  \begin{tikzpicture}
    \node[frameb,font=\small] (a) {all files};
    \node[frameb,font=\small,right=1.4cm of a] (b) {model};
    \draw[arrA] (a) -- (b);
  \end{tikzpicture}
  \end{center}
  \vspace{0.4em}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \group{Strengths}
      {\small
      Simple: no index, no pipeline.\\[0.4em]
      The model reads every source in the author's own words.\\[0.4em]
      Works while the material fits comfortably in the window.\par}
    \column{0.48\textwidth}
      \group{Limits}
      {\small
      The window caps the corpus.\\[0.4em]
      Every request pays for every token.\\[0.4em]
      Irrelevant pages dilute attention.\\[0.4em]
      Every update means sending everything again.\par}
  \end{columns}
  \takeaway{Fine for one paper. Not for a research agenda.}
\end{frame}

  \vspace{0.35em}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      {\small
      \textbf{\color{ink}Lexical}: matches words (BM25). Good for names, codes, variable labels.\\[0.3em]
      \textbf{\color{ink}Semantic}: matches meaning through embeddings. Good for paraphrase.\\[0.3em]
      \textbf{\color{ink}Hybrid}: runs both and merges the rankings.\par}
    \column{0.48\textwidth}
      \renewcommand{\arraystretch}{1.15}
      {\scriptsize
      \begin{tabularx}{\linewidth}{>{\bfseries\color{ink}}l >{\leavevmode\color{dark}}X}
      \toprule
      Failure & What happened\\
      \midrule
      Wrong chunk & the split cut the key sentence\\
      Missed source & the right document never ranks\\
      Incomplete context & the answer needs ten passages; k is five\\
      Unfaithful synthesis & the model writes past what the passages say\\
      \bottomrule
      \end{tabularx}}
  \end{columns}
  \source{\citet{lewis_2020_rag}; \citet{robertson_2009_bm25}.}

\begin{frame}{Ingest, query, lint}
  \begin{columns}[T,onlytextwidth]
    \column{0.30\textwidth}
      \group{Ingest}
      {\small Read one new source. Write its summary page. Update every page it touches. Log what changed.\par}
    \column{0.30\textwidth}
      \group{Query}
      {\small Ask the wiki, not the raw pile. A good answer can be filed back as a page.\par}
    \column{0.30\textwidth}
      \group{Lint}
      {\small Look for contradictions, stale claims, orphan pages, missing links, concepts without a page.\par}
  \end{columns}
  \vspace{0.45em}
  \begin{center}
  \begin{tikzpicture}
    \node[accb,font=\small,align=center,text width=0.92\textwidth] {Promotion: a result checked in a session becomes a page, a decision, or a skill.};
  \end{tikzpicture}
  \end{center}
  \vspace{0.35em}
  \group{One paper, five edits}
  {\small summary page (new); staggered DiD concept page (estimator added); employment entity page (new estimate); open questions (conflict with an earlier result flagged); index (two new rows).\par}
  \source{\citet{karpathy_2026_llmwiki}.}
\end{frame}

\begin{frame}{Comparing architectures, and firms}
  \renewcommand{\arraystretch}{1.08}
  {\fontsize{6.2}{7.4}\selectfont
  \begin{tabularx}{\linewidth}{>{\bfseries\color{ink}}l l l l l l}
  \toprule
  Design & Setup & Keeping current & Retrieval & Provenance & Reuse\\
  \midrule
  Full prompt & none & resend everything & complete, if it fits & the source itself & none\\
  RAG & index pipeline & re-index changed files & depends on chunking & passage & low\\
  Knowledge base & curation time & manual review & high on covered topics & entry & high\\
  GraphRAG & extraction, many tokens & re-extract & good on global questions & entity, indirect & medium\\
  LLM Wiki & a folder and a schema & the agent, at each ingest & good for synthesis & links to raw sources & high\\
  Hybrid & highest & both & best, if maintained & both & high\\
  \bottomrule
  \end{tabularx}}
  \vspace{0.15em}
  {\scriptsize\color{muted} Qualitative comparison by the author.\par}
  \vspace{0.25em}
  \group{In firms}
  {\small Glean and Microsoft 365 Copilot index company documents and answer within each user's existing permissions. Meta has described an internal organisational second brain.\par}
  \vspace{0.25em}
  {\small The hard part is governance: who may see what, stale pages that look authoritative, memory that outlives the reason it was written.\par}
  \source{\citet{glean_product}; \citet{m365_copilot}; \citet{meta_2026_secondbrain}.}
\end{frame}
