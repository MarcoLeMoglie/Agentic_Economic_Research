\begin{frame}{The problem with ordinary chatbots}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \group{Six months of a paper in chat windows}
      {\scriptsize\color{muted} Illustrative example.\par}
      \vspace{0.35em}
      {\small
      Week 2: you explain the sample and the treatment again.\\[0.35em]
      Week 5: the agent clusters by province. You had decided on municipality.\\[0.35em]
      Week 9: a new chat. Why 2020 was dropped is gone.\\[0.35em]
      Week 14: two chats give two different rules for outliers.\\[0.35em]
      Week 20: the coauthor's instructions sit in a thread you cannot find.\par}
    \column{0.50\textwidth}
      \renewcommand{\arraystretch}{1.25}
      \begin{tabularx}{\linewidth}{>{\bfseries\color{ink}}l >{\small\leavevmode\color{dark}}X}
      \toprule
      \color{muted}Problem & \color{muted}What it costs\\
      \midrule
      Repeated context & tokens and time at every start\\
      Lost decisions & the same choice made twice, differently\\
      Hard restarts & the first twenty minutes spent rebuilding\\
      Inconsistent instructions & outputs that contradict each other\\
      Fragmentation & no single place to check\\
      \bottomrule
      \end{tabularx}
  \end{columns}
  \takeaway{A chat log records what was said. It is a poor record of what was decided.}
\end{frame}

\begin{frame}{Persistent memory}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \renewcommand{\arraystretch}{1.18}
      \begin{tabularx}{\linewidth}{>{\bfseries\color{ink}}l >{\small\leavevmode\color{dark}}X}
      \toprule
      \color{muted}What to keep & \color{muted}In one line\\
      \midrule
      State & where the project stands today\\
      Decisions & what was chosen, when, and why\\
      Constraints & what must not change\\
      Sources & where each number comes from\\
      Solved errors & the bug and the fix that worked\\
      Next actions & the first thing to do next time\\
      Procedures & steps you repeat, written once\\
      \bottomrule
      \end{tabularx}
    \column{0.48\textwidth}
      \defn{Chat history}{Everything said, in order. Long, and mostly noise for the next session.}{}
      \vspace{0.45em}
      \defn{Document archive}{Papers, data, drafts. The evidence, not the summary.}{}
      \vspace{0.45em}
      \defn{Persistent memory}{Short files, kept current by you or the agent, read at every start.}{}
  \end{columns}
  \takeaway{Memory is what you would tell a coauthor who joins tomorrow.}
\end{frame}

\begin{frame}{Fewer tokens, better context}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \begin{tikzpicture}
        \fill[faint] (0,1.15) rectangle (5.15,1.48);
        \node[anchor=west,font=\scriptsize,text=ink] at (0,1.65) {Load everything};
        \node[anchor=west,font=\scriptsize,text=muted] at (5.25,1.31) {100{,}000};
        \fill[accent] (0,0.25) rectangle (0.41,0.58);
        \node[anchor=west,font=\scriptsize,text=ink] at (0,0.75) {Load what the task needs};
        \node[anchor=west,font=\scriptsize,text=muted] at (0.55,0.41) {8{,}000};
      \end{tikzpicture}
      \vspace{0.45em}
      {\small\bfseries\color{ink} 92\% fewer input tokens for this request.\par}
      \vspace{0.2em}
      {\scriptsize\color{muted} Illustrative numbers, not a measured benchmark.\par}
    \column{0.50\textwidth}
      \renewcommand{\arraystretch}{1.2}
      {\small
      \begin{tabularx}{\linewidth}{>{\bfseries\color{ink}}l >{\leavevmode\color{dark}}X}
      Progressive disclosure & names and descriptions first, full text on demand\\
      Selective loading & one project's files, not the whole vault\\
      Retrieval & the few passages that match the question\\
      Prompt caching & a repeated prefix billed at a discount\\
      Compaction & old turns replaced by a summary\\
      \end{tabularx}\par}
  \end{columns}
  \vspace{0.45em}
  {\small Savings per request are not total savings. Writing the notes, building an index, and keeping both current also cost tokens and time.\par}
  \source{\citet{anthropic_prompt_caching}; \citet{openai_prompt_caching}.}
\end{frame}

\begin{frame}{Portable skills}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \filebox{assets/lesson2/files/skill_excerpt.md}{Opening of the demo skill \texttt{check-synthetic-panel}. The agent sees the name and description first.}
    \column{0.44\textwidth}
      {\small
      At start, the agent sees each skill's name and description: a few dozen tokens.\\[0.45em]
      When a request matches, it reads the full \texttt{SKILL.md}.\\[0.45em]
      Scripts, templates and references load only if the steps call for them.\\[0.45em]
      The same folder works in any harness that follows the Agent Skills format.\par}
      \vspace{0.4em}
      {\scriptsize\color{muted} The saving comes from loading in stages. Pasting a folder of procedures into every prompt saves nothing.\par}
  \end{columns}
  \source{\citet{agentskills_spec}; \citet{claude_skills}; \citet{codex_skills}.}
\end{frame}
