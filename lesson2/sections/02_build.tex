\begin{frame}{Build your first brain}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \filebox{assets/lesson2/files/min_tree.txt}{A first vault. Six files are enough.}
    \column{0.50\textwidth}
      \renewcommand{\arraystretch}{1.18}
      {\small
      \begin{tabularx}{\linewidth}{>{\ttfamily\bfseries\color{ink}}l >{\leavevmode\color{dark}}X}
      README.md & what this vault is, for you and for the agent\\
      INDEX.md & the map; read first\\
      PROJECT\_BRIEF & question, scope, people; changes rarely\\
      CURRENT\_STATE & where things stand today\\
      NEXT\_ACTIONS & the next three things to do\\
      DECISIONS & date, decision, reason\\
      \end{tabularx}}
  \end{columns}
  \takeaway{Six files are enough for the first month. Add structure when a file gets too long.}
\end{frame}

\begin{frame}{Connect an AI agent}
  \begin{columns}[T,onlytextwidth]
    \column{0.28\textwidth}
      \begin{tikzpicture}
        \node[accb,font=\small\bfseries,text=ink] (a) at (0,1.2) {Agent};
        \node[frameb,font=\small\bfseries,text=ink] (v) at (0,-1.2) {Vault};
        \draw[arrA] (a) -- (v);
      \end{tikzpicture}
    \column{0.68\textwidth}
      {\small
      \textbf{\color{ink}Folder access}: open the vault as the workspace, or add it as an extra folder.\\[0.28em]
      \textbf{\color{ink}Permissions}: read freely; ask before writing; nothing outside the vault.\\[0.28em]
      \textbf{\color{ink}Instructions}: \texttt{AGENTS.md} at the root points to \texttt{BOOTSTRAP.md}.\\[0.28em]
      \textbf{\color{ink}Bootstrap}: read INDEX, then only the project you name.\\[0.28em]
      \textbf{\color{ink}Canonical memory}: the Markdown files. Anything else is support.\\[0.28em]
      \textbf{\color{ink}Controlled writes}: the agent proposes edits; Git shows the diff.\par}
  \end{columns}
  \vspace{0.4em}
  \defn{Reading is not remembering}{The agent remembers only what someone wrote back into the files.}{}
\end{frame}

\begin{frame}{Git, GitHub and Obsidian}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \shoth{obsidian-git-history.png}{2.35cm}{Git log, all branches. Audit on \texttt{final-verified}.}
    \column{0.48\textwidth}
      \shoth{obsidian-git-diff.png}{2.35cm}{Diff of the decisions file after the audit.}
  \end{columns}
  \vspace{0.2em}
  \renewcommand{\arraystretch}{1.12}
  {\small
  \begin{tabularx}{\linewidth}{>{\bfseries\color{ink}}l >{\leavevmode\color{dark}}X}
  Dropbox & syncs files across devices; keeps versions, without messages\\
  Git & records each change with a message: diff, history, rollback\\
  GitHub & hosts the repository; make it private for notes\\
  Obsidian Git & commits and pushes from inside Obsidian\\
  \end{tabularx}}
  \vspace{0.25em}
  {\scriptsize\color{muted} \texttt{.gitignore} keeps secrets, caches and large data out. Two machines editing the same note create a conflict you resolve by hand.\par}
  \source{\citet{progit}; \citet{github_visibility}; \citet{obsidian_git}.}
\end{frame}
