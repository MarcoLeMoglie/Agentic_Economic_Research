\begin{frame}[t]{Part 3: how I built and wired the Brain}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \textbf{Build \& folder structure}
      \begin{itemize}\setlength\itemsep{0.08em}
        \item Obsidian vault = \textbf{Markdown source of truth} (not blind RAG).
        \item \texttt{01\_GENERAL/} --- agent protocols, skills, integrations, scripts.
        \item \texttt{02\_AREAS/} --- Teaching, Research, Other; one folder per project.
        \item Per project: 8 canonical files + \texttt{wiki/} + \texttt{raw/chats/}.
        \item Populate via \texttt{create\_project.py}, session closeout, LLM Wiki ingest.
      \end{itemize}
      \textbf{Integrations (support layers)}
      \begin{itemize}\setlength\itemsep{0.06em}
        \item \textbf{Anti-sloppiness} --- Slopless on formal prose.
        \item \textbf{Headroom} ---context compression (tokens, long logs).
        \item \textbf{Memanto} --- typed semantic recall; 
        \item \textbf{code-review-graph} --- impact-aware code review.
        \item \textbf{Econoclast} --- adversarial referee on empirical papers.
        \item \textbf{Style / taste} --- voice profile + archived examples.
      \end{itemize}
    \column{0.48\textwidth}
      \textbf{Protocols \& skills}
      \begin{itemize}\setlength\itemsep{0.08em}
        \item \textbf{LLM Wiki} --- promote stable notes into project \texttt{wiki/}.
        \item \textbf{Caveman} --- terse agent mode; saves tokens, keeps substance.
        
      \end{itemize}
      \textbf{``Open brain session''}
      \begin{itemize}\setlength\itemsep{0.06em}
        \item Read global + project memory (\texttt{INDEX}, \texttt{CURRENT\_STATE}, \ldots).
        \item Inspect live files; summarize state + plan --- \emph{before} any edits.
      \end{itemize}
      \textbf{``Close brain session''}
      \begin{itemize}\setlength\itemsep{0.06em}
        \item Update \texttt{CURRENT\_STATE}, \texttt{NEXT\_ACTIONS}, \texttt{DECISIONS}, \texttt{LOG}.
        \item Promote durable facts to \texttt{AGENT\_MEMORY} / \texttt{wiki/}; optional Memanto sync.
      \end{itemize}
  \end{columns}
\end{frame}
