  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \textbf{We will show}
      \begin{itemize}\setlength\itemsep{0.35em}
        \item Folder organization: General Areas, projects, wiki vs.\ raw notes.
        \item Brain bootstrap: opening and closing a session
        \item Tools, plugins, protocols, skills
      \end{itemize}
    \column{0.48\textwidth}
      \textbf{Takeaway for you}
      \begin{itemize}\setlength\itemsep{0.35em}
        \item Start small: one project folder, eight canonical files.
        \item Write decisions in Markdown, not only in chat.
        \item Treat the vault as the restart point every session.
      \end{itemize}
  \end{columns}
\end{frame}
