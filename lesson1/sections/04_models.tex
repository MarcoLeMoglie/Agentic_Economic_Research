% -------- Open weights, open source, local --------
\begin{frame}{Closed weights and open weights}
  \begin{columns}[c,onlytextwidth]
    \column{0.48\textwidth}
      \defn{Closed weights}{\rev{You call the model through an API. The lab keeps the parameters. Fine-tuning them is not available to you.}}{}
      \vspace{0.7em}
      \defn{Open weights}{\rev{You download the parameters and can fine-tune them for your own task, under the model's licence.}}{DeepSeek, GLM, Qwen, MiMo, Kimi K3, gpt-oss, Gemma.}
    \column{0.46\textwidth}
      {\small
      \group{Local pays off when}
      {\color{muted}data cannot leave the machine, or the same small task runs thousands of times.\par}
      \vspace{0.6em}
      \group{Local hurts when}
      {\color{muted}the task needs frontier reasoning or long agent loops on a laptop. Tools: Ollama, LM Studio.\par}}
  \end{columns}
\end{frame}
