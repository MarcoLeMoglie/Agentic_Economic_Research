% -------- Chatbot vs agent --------
\begin{frame}{Chatbot vs agent}
  \vspace{0.6em}
  \begin{center}
  \renewcommand{\arraystretch}{1.25}
  \begin{tabularx}{0.86\linewidth}{>{\bfseries\color{ink}}l X X}
  \toprule
   & \leavevmode\color{muted}Chatbot & \leavevmode\color{muted}Agent\\
  \midrule
  Who runs the loop & You & The harness\\
  Unit of delegation & An answer & A task\\
  Files & Pasted or attached & Read, edited, created\\
  Failure mode & A wrong \emph{answer} & A wrong \emph{action}\\
  Your role & Reader & Supervisor\\
  \bottomrule
  \end{tabularx}
  \end{center}
  \takeaway{Different failure mode, different control.}
\end{frame}

% -------- The loop, in a real trace --------
\begin{frame}{The loop, in a real trace}
  \begin{columns}[T,onlytextwidth]
    \column{0.34\textwidth}
      \vspace{0.15em}
      {\scriptsize
      \group{\textcolor{ink}{Observe}}{\color{ink}reads the do-file\par}
      \vspace{0.35em}
      \group{\textcolor{accent}{Act}}{\color{accent}runs Stata.\par}
      \vspace{0.35em}
      \group{\textcolor{stepR}{Observe}}{\textcolor{stepR}{reads the log. The diff is the typo: \texttt{ln(prise)}.}\par}
      \vspace{0.35em}
      \group{\textcolor{stepE}{Act}}{\textcolor{stepE}{replaces that one line. Minus is the old line, plus is the new one.}\par}
      \vspace{0.35em}
      \group{\textcolor{good}{Verify}}{\textcolor{good}{runs it again, and the table is there.}\par}}
    \column{0.63\textwidth}
      \begin{tcolorbox}[l1base,colback=white,borderline={0.4pt}{0pt}{hair},left=0.45em,right=0.3em,top=0.3em,bottom=0.3em]
        \ttfamily\fontsize{5.6}{6.7}\selectfont
        {\color{ink}> build · deepseek-v4.1-flash-free\\
        → Read code/analysis.do}\\[0.15em]
        {\color{accent}\$ .../stata-mp -b do code/analysis.do; echo "EXIT:\$?"\\
        EXIT:0}\\[0.15em]
        {\color{stepR}→ Read analysis.log\\
        -gen lprice = ln(prise)}\\[0.15em]
        {\color{stepE}+gen lprice = ln(price)}\\[0.15em]
        {\color{good}\$ .../stata-mp -b do code/analysis.do; echo "EXIT:\$?"\\
        EXIT:0\\
        (output written to output/table1.tex)}
      \end{tcolorbox}
  \end{columns}
\end{frame}
